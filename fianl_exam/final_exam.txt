parent = 1; child = 2;
while (child <= *n) {
    // 두 자식 중 큰 쪽 선택
    if (child < *n && heap[child].key < heap[child+1].key)
        child++;
    if (temp.key >= heap[child].key) break;  // 자리 확정
    heap[parent] = heap[child];  // 자식을 위로 올림
    parent = child;
    child *= 2;
}

    int temp = parent[i] + parent[j]; // 둘 다 음수라 합치면 더 큰 음수
    if (parent[i] > parent[j]) {  // i가 노드 수 적으면

        parent[j] = temp;

        parent[i] = temp;
    }
}

int collapsingFind(int i) {
    int root, trail, lead;
    for (root = i; parent[root] >= 0; root = parent[root]); // 1단계 root 찾기
    // 2단계: 경로상 모든 노드 → root 직접 자식으로
    for (trail = i; trail != root; trail = lead) {
        lead = parent[trail];      // 다음 노드 저장
        parent[trail] = root;      // root 직접 자식으로
    }
    return root;
}
