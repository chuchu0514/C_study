① 그래프 (BFS/DFS/cycle/CC)

구조체는 연결리스트랑 똑같음 (data + next), 그걸 정점 개수만큼 배열로
node** g = 정점마다 리스트 하나. undirected면 양방향 insert
BFS = 큐(배열) + 반복, DFS = 재귀
visited 배열 필수. cycle은 "방문했는데 부모 아님", CC는 "미방문 정점마다 DFS 새로 시작"

② BST (preorder/subtree/LCA/비교)

구조체는 자식 2개 (data + left + right)
삽입·순회·비교 전부 재귀가 기본 (작으면 왼쪽 / 크면 오른쪽)
LCA만 예외적으로 반복으로 짜기 편함 (root에서 내려오며 갈라지는 지점)
핵심 감각: "한 노드 처리 + 왼쪽 재귀 + 오른쪽 재귀"

③ 정렬 (radix/heap)

radix = counting sort를 자릿수만큼 반복, 무조건 뒤에서부터 채워야 stable
heap = 1차원 배열, parent=i/2 / child=2i, 2i+1
둘 다 결과 거꾸로 출력하면 내림차순

## 1-1. 자료구조와 입력 구성 (linked adjacency list, undirected)

```c
typedef struct node {
    int data;
    struct node* next;
} node;

// 정렬 삽입: "작은 번호부터 방문" 제약이 있을 때 사용한다.
// (단순 head 삽입을 하면 리스트가 역순으로 쌓여 큰 번호부터 나오므로 주의)
void insert(node** head, int data){
    node* newNode = (node*)malloc(sizeof(node));
    newNode->data = data;

    // 리스트가 비었거나, 첫 노드보다 작으면 맨 앞에 삽입
    if(*head == NULL || (*head)->data > data){
        newNode->next = *head;
        *head = newNode;
        return;
    }
    // data보다 작은 노드들을 지나친 뒤, 그 자리에 끼워넣는다.
    node* current = *head;
    while(current->next != NULL && current->next->data < data){
        current = current->next;
    }
    newNode->next = current->next;
    current->next = newNode;
}

## 1-2. BFS — 거리 제한형 (예: 친구 + 친구의 친구 = 거리 2까지)

```c
void bfs(node** adjacencyList, int start, int* visited, int maxDistance){
    int queue[100][2];          // [0] = 정점 번호, [1] = 시작점으로부터의 거리
    int front = 0, rear = 0;

    visited[start] = 1;
    queue[rear][0] = start;
    queue[rear][1] = 0;
    rear++;

    while(front < rear){
        int vertex   = queue[front][0];
        int distance = queue[front][1];
        front++;

        // 거리 제한: maxDistance에 도달한 노드는 더 이상 확장하지 않는다.
        if(distance == maxDistance) continue;

        for(node* current = adjacencyList[vertex]; current != NULL; current = current->next){
            if(visited[current->data] == 0){
                visited[current->data] = 1;
                printf("%d ", current->data);   // 시작점 자신은 제외하고 출력됨
                queue[rear][0] = current->data;
                queue[rear][1] = distance + 1;
                rear++;
            }
        }
    }
}
```

// cycle 탐지: 이미 방문한 노드인데 그게 부모가 아니면 cycle이 존재한다.
int hasCycle = 0;
void dfsCycle(node** adjacencyList, int vertex, int parent, int* visited){
    visited[vertex] = 1;
    for(node* current = adjacencyList[vertex]; current != NULL; current = current->next){
        if(visited[current->data] == 0){
            dfsCycle(adjacencyList, current->data, vertex, visited);
        }
        else if(current->data != parent){
            hasCycle = 1;     // 방문했는데 부모가 아님 → cycle
        }
    }
}

int color[100];

int isBipartite(node** adjacencyList, int n){
    for(int i = 0; i < n; i++) color[i] = 0; 색 일단 다 0으로 초기화 색이 칠해지지 않았음 
    for(int s = 0; s < n; s++){
        if(color[s] != 0) continue; // 이미 칠해졌으면(다른 BFS가 처리함) 건너뜀
        int queue[100], front = 0, rear = 0; // 안 칠해졌으면 여기서 새 BFS 시작
        color[s] = 1;
        queue[rear++] = s;
        while(front < rear){
            int v = queue[front++];
            for(node* c = adjacencyList[v]; c; c = c->next){
                if(color[c->data] == 0){
                    color[c->data] = 3 - color[v];   // 1↔2 반대 색
                    queue[rear++] = c->data;
                }
                else if(color[c->data] == color[v]){
                    return 0;                         // 같은 색 이웃 → 이분 아님
                }
            }
        }
    }
    return 1;
}

treeNode* lca(treeNode* root, int a, int b){
    treeNode* temp = root;
    while(temp != NULL){
        if(a < temp->data && b < temp->data){
            temp = temp->left;        // 둘 다 temp보다 작으면 왼쪽으로
        }

        else {
            return temp;              // 한쪽은 작고 한쪽은 크거나 같음 → 여기가 LCA
        }
    }
    return NULL;
}

# ③ 정렬 (LSD radix sort / min heap)


## ③-A. LSD Radix Sort (교재 방식: link 연결 + front/rear bin)

### digit 함수 — 어떤 수의 특정 자릿수 추출

```c
// number에서 오른쪽 기준 pos번째 자릿수 (pos=1이면 1의 자리, 2면 10의 자리 ...)
int digit(int number, int pos, int radix){
    for(int i = 1; i < pos; i++){
        number /= radix;
    }
    return number % radix;
}
```

### radixSort — 교재 link 방식

```c
// a[1..n]에 데이터, link[1..n]은 보조 배열, d=자릿수, radix=10
// 정렬이 끝나면 first가 가장 작은 원소의 인덱스, link를 따라가면 오름차순
int radixSort(int* a, int* link, int n, int d, int radix){
    int front[10], rear[10];   // radix개의 bin (여기선 10)
    int first, current, last;

    // 초기 체인: 1 -> 2 -> 3 -> ... -> n -> 0(끝)
    first = 1;
    for(int i = 1; i < n; i++) link[i] = i + 1;
    link[n] = 0;

    // 가장 낮은 자리(pos=1)부터 가장 높은 자리(pos=d)까지 반복
    for(int pos = 1; pos <= d; pos++){

        // 1) bin 비우기
        for(int b = 0; b < radix; b++){
            front[b] = 0;
            rear[b] = 0;
        }

        // 2) distribution: 체인을 따라가며 각 원소를 해당 bin에 매단다
        for(current = first; current != 0; current = link[current]){
            int b = digit(a[current], pos, radix);
            if(front[b] == 0){
                front[b] = current;          // bin이 비었으면 시작점
            } else {
                link[rear[b]] = current;     // 아니면 그 bin의 끝에 이어붙임
            }
            rear[b] = current;               // 끝을 갱신 (rear에 추가 = stable)
        }

        // 3) merging: bin 0번부터 9번까지 순서대로 다시 하나의 체인으로 연결
        first = 0;
        last = 0;
        for(int b = 0; b < radix; b++){
            if(front[b] != 0){               // 비어있지 않은 bin만
                if(first == 0){
                    first = front[b];        // 첫 bin의 시작이 전체 시작
                } else {
                    link[last] = front[b];   // 이전 bin 끝과 이 bin 시작을 연결
                }
                last = rear[b];
            }
        }
        if(last != 0) link[last] = 0;        // 마지막 원소의 link는 0(끝)
    }

    return first;   // 가장 작은 원소의 인덱스
}
```

### 자릿수 d 자동 계산 (입력 범위 모를 때)

```c
int maxValue = a[1];
for(int i = 2; i <= n; i++) if(a[i] > maxValue) maxValue = a[i];

**radix 함정 정리**
- 데이터 배열은 **1-indexed** (교재 관례). `a[1..n]`, `link[n]=0`이 끝 표시.
- distribution에서 **rear에 매달기** = stable. 이 순서 바뀌면 안 됨.
- merging은 **bin 0→9 순서**로. 이 순서가 오름차순을 만든다.
- nonincreasing은 오름차순 정렬 후 **거꾸로 출력** (또는 merging을 9→0으로).
- `digit`의 pos는 1부터 시작(1의 자리). LSD니까 pos=1부터 d까지.

## ③-B. Heap Sort (교재 방식: adjust / heapSort)

> 교재 heap sort는 `adjust`(down-heap)로 max heap을 만들고,
> 루트와 마지막을 교환하며 크기를 줄여간다. **배열 1-indexed.**

```c
// a[1..n]을 정렬. root=i, 자식 = 2i, 2i+1
// adjust: root 위치의 원소를 아래로 내려보내 heap 성질 회복 (max heap 기준)
void adjust(int* a, int root, int n){
    int child = 2 * root;
    int rootValue = a[root];
    while(child <= n){
        // 두 자식 중 큰 쪽 선택
        if(child < n && a[child] < a[child + 1]){
            child++;
        }
        // 부모가 더 크거나 같으면 멈춤
        if(rootValue >= a[child]){
            break;
        }
        a[child / 2] = a[child];   // 자식을 부모 자리로 끌어올림
        child *= 2;
    }
    a[child / 2] = rootValue;       // 최종 자리에 원래 root 값 넣기
}

void heapSort(int* a, int n){
    // 1) 초기 max heap 구성: 마지막 부모(n/2)부터 1까지 adjust
    for(int i = n / 2; i >= 1; i--){
        adjust(a, i, n);
    }
    // 2) 루트(최대값)를 맨 뒤로 보내고, heap 크기를 줄여 다시 adjust
    for(int i = n - 1; i >= 1; i--){
        int temp = a[1]; a[1] = a[i + 1]; a[i + 1] = temp;  // 루트 <-> 끝 교환
        adjust(a, 1, i);                                    // 줄어든 heap 재정렬
    }
    // 결과: 오름차순 (max heap이라 큰 게 뒤로 쌓임)
}
```

**heap 함정 정리**
- 배열 **1-indexed**. `a[1]`이 루트.
- max heap → heap sort 결과는 **오름차순**. 내림차순 출력은 거꾸로.
- `adjust`에서 자식 비교(`a[child] < a[child+1]`) 방향이 max/min을 결정.
  - PPT 예시가 min heap이면 시험은 max heap으로 뒤집어 낼 수 있음 → 부등호 방향 주의.
- 교재의 `adjust`/`heapSort` 구조를 그대로 따라야 함. 다른 방식이면 0점 위험.
